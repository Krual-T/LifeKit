(

    printf '%s\n' '目标库已存在，保留原内容；恢复前请先移走旧库。' >&2

  git -C "$task_pm_source" archive FETCH_HEAD skills docs scripts README.md LICENSE |
    tar -x -C "$task_pm_source/library"
  mv "$task_pm_source/library" "$task_pm_library"
  printf '已恢复参考库；临时 Git 下载目录：%s\n' "$task_pm_source"
